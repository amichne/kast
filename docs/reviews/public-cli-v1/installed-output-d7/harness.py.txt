from pathlib import Path
from dataclasses import dataclass,asdict
import json,os,signal,subprocess
import measure as m
out=m.ROOT/'evidence/installed-output-cancellation';out.mkdir(exist_ok=False)
argv=[str(m.HOME/'.local/bin/kast'),'describe','query','--section','output']
jcmd='/Users/amichne/Applications/IntelliJ IDEA.app/Contents/jbr/Contents/Home/bin/jcmd'
with (out/'control.stderr').open('wb') as err:
 control=subprocess.Popen(argv,cwd=m.WORKSPACE,env=m.ENVIRONMENT,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=err)
 try:control_bytes,_=control.communicate(timeout=60)
 except subprocess.TimeoutExpired:control.terminate();control.wait(timeout=30);raise
assert control.returncode==0
assert control_bytes.count(b'\n')==1
json.loads(control_bytes)
(out/'control.stdout').write_bytes(control_bytes)
with (out/'signal.stderr').open('wb') as err:
 child=subprocess.Popen(argv,cwd=m.WORKSPACE,env=m.ENVIRONMENT,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=err)
 try:
  jvm_pid=None
  for index in range(100):
   processes=subprocess.run(['/usr/bin/pgrep','-P',str(child.pid)],stdout=subprocess.PIPE,check=False)
   pids=processes.stdout.decode().split()
   if len(pids)!=1:continue
   jvm_pid=int(pids[0])
   dump=subprocess.run([jcmd,str(jvm_pid),'Thread.print'],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,timeout=15,check=False)
   text=dump.stdout.decode();(out/f'before-{index}.txt').write_text(text)
   worker=text.split('"kast-public-query"',1)[-1].split('\n\n',1)[0]
   if 'Native Method' in worker and ('write0' in worker or 'writeBytes' in worker) and 'PublicInvocationOwner.publish' in worker and 'PublicCliInvocationKt.joinPublicWorker' in text:break
  else:raise AssertionError('No blocked-writer precondition')
  assert jvm_pid is not None
  child.send_signal(signal.SIGINT)
  status=child.wait(timeout=10)
  assert status==130,status
  try:os.kill(jvm_pid,0)
  except ProcessLookupError:pass
  else:raise AssertionError('Owned JVM remains after native exit')
  assert child.stdout is not None
  partial=child.stdout.read();(out/'signal.stdout').write_bytes(partial)
 finally:
  if child.poll() is None:
   child.terminate()
   try:child.wait(timeout=10)
   except subprocess.TimeoutExpired:child.kill();child.wait(timeout=10)
@dataclass(frozen=True)
class Result:
 argv:list[str]
 nativePid:int
 jvmPid:int
 controlExit:int
 controlBytes:int
 signalExit:int
 partialBytesAfterSettlement:int
 readerBeforeSettlement:str
 ownedJvmReaped:bool
 candidate:dict
result=Result(argv,child.pid,jvm_pid,control.returncode,len(control_bytes),status,len(partial),'NO_READS',True,json.loads((m.ROOT/'final-candidate.json').read_text()))
(out/'result.json').write_text(json.dumps(asdict(result),indent=2)+'\n')
print('installed native/JBR blocked stdout SIGINT PASS',status,len(partial))
